Write a c program to implement round robin scheduling where take process id, burst time, arrival time as user input

#include <stdio.h>

int main() {
    int n, tq;
    int bt[20], rem_bt[20], wt[20], tat[20], at[20];
    int i, time = 0, count = 0;

    printf("Enter the number of processes: ");
    scanf("%d", &n);

    printf("Enter the time quantum (in sec): ");
    scanf("%d", &tq);

    printf("Enter the arrival times for all processes:\n");
    for (i = 0; i < n; i++) {
        printf("P%d: ", i + 1);
        scanf("%d", &at[i]);
    }

    printf("Enter the burst times for all processes:\n");
    for (i = 0; i < n; i++) {
        printf("P%d: ", i + 1);
        scanf("%d", &bt[i]);
        rem_bt[i] = bt[i];
        wt[i] = 0;
        tat[i] = 0;
    }

    int done;
    do {
        done = 1;
        for (i = 0; i < n; i++) {
            // Execute only if process has arrived and has remaining burst time
            if (rem_bt[i] > 0 && at[i] <= time) {
                done = 0;
                if (rem_bt[i] > tq) {
                    time += tq;
                    rem_bt[i] -= tq;
                } else {
                    time += rem_bt[i];
                    wt[i] = time - bt[i] - at[i];  // Waiting time considering arrival
                    rem_bt[i] = 0;
                }            }
        }
        // If no process has arrived yet, increment time
        int all_not_arrived = 1;
        for (i = 0; i < n; i++) {
            if (rem_bt[i] > 0 && at[i] <= time) {
                all_not_arrived = 0;
                break;
            }
        }
        if (all_not_arrived) time++;
    } while (done == 0);

    for (i = 0; i < n; i++)
        tat[i] = bt[i] + wt[i];

    printf("\nProcess\tArrival Time\tBurst Time\tWaiting Time\tTurnaround Time\n");
    for (i = 0; i < n; i++) {
        printf("P%d\t%d\t\t%d\t\t%d\t\t%d\n", i + 1, at[i], bt[i], wt[i], tat[i]);
    }

    float avg_wt = 0, avg_tat = 0;
    for (i = 0; i < n; i++) {
        avg_wt += wt[i];
        avg_tat += tat[i];
    }

    avg_wt /= n;
    avg_tat /= n;

    printf("\nAverage Waiting Time = %.2f", avg_wt);
    printf("\nAverage Turnaround Time = %.2f\n", avg_tat);

    return 0;
}
