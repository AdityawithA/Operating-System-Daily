//Write a shell program to take input from user two ranges and print total number of even numbers and also print the odd number.

echo -n "Enter the starting number: "
read start

echo -n "Enter the ending number: "
read end

even_count=0

echo "Odd numbers in the range $start to $end are:"
for (( i=start; i<=end; i++ ))
do
  if (( i % 2 == 0 ))
  then
    even_count=$((even_count + 1))
  else
    echo "$i"
  fi
done
